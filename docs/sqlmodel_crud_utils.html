584    id_str_list: list[str | int],
585    data: dict,
586    session_inst: Session,
587    model: type[SQLModel],
588    pk_field: str = "id",
589    chunk_size: int = 500,
590) -> tuple[bool, int]:
591    """
592    Applies the same column values to every row whose primary key is within
593    the provided list and commits once. IDs with no matching row are ignored.
594
595    This is a Core ``UPDATE``: ORM-level hooks and Python-side defaults do
596    not run, so ``AuditMixin.updated_at``/``updated_by`` are not refreshed
597    unless the caller includes them in ``data`` (column-level ``onupdate``
598    defaults, such as ``TimestampMixin.updated_at``, do fire). The ID list is
599    split into ``chunk_size`` batches. One ``UPDATE ... WHERE pk IN (...)``
600    statement is sent per batch (never per row), and all batches share one
601    transaction.
602
603    :param id_str_list: List of primary key values to update.
604    :param data: Mapping of column name to the new value for every row. Must
605        be non-empty and may not name a primary-key column.
606    :param session_inst: SQLAlchemy Session instance.
607    :param model: SQLModel class representing the table.
608    :param pk_field: Primary-key column to match against (default: "id").
609    :param chunk_size: Maximum IDs per UPDATE statement (default: 500).
610    :return: Tuple[bool, int]: ``True`` and the number of rows matched by the
611        update. The count is 0 when nothing matched or the list was empty.
612    :raises ValueError: If ``pk_field`` is not a primary-key column,
613        ``data`` is empty or names an unknown or primary-key column, or
614        ``chunk_size`` is not positive.
615    :raises Exception: Any database error, after the transaction has been
616        rolled back and the error logged.
617    """
618    validate_primary_key_field(model, pk_field)
619    validate_update_columns(model, data)
620    updated = 0
621    try:
622        for chunk in chunked(id_str_list, chunk_size):
623            stmnt = (
624                update(model)
625                .where(getattr(model, pk_field).in_(chunk))
626                .values(**data)
627            )
628            result = session_inst.exec(stmnt)
629            updated += max(result.rowcount, 0)
630        session_inst.commit()
631    except Exception as e:
632        session_inst.rollback()
633        logger.error(f"Failed to bulk update rows: {type(e)}, {e}")
634        raise
635
636    invalidate_written(session_inst, model)
637    return True, updated

678def bulk_upsert_mappings(
679    payload: list,
680    session_inst: Session,
681    model: type[SQLModel],
682    pk_fields: list[str] | None = None,
683):
684    """
685    Insert or update a batch of rows in a single ``INSERT ... ON CONFLICT DO
686    UPDATE ... RETURNING`` statement.
687
688    The ``RETURNING`` rows are fully read (``.all()``) before ``commit()``.
689    SQLite refuses to commit while a result cursor is still unread
690    ("cannot commit transaction - SQL statements in progress"); Postgres
691    tolerates it. The upsert is also executed exactly once rather than once
692    bare and once more to fetch the rows.
693
694    Returned instances are expired by ``commit()`` when the session uses
695    ``expire_on_commit=True`` (the default), so read their attributes while
696    the session is open or create the session with ``expire_on_commit=False``.
697
698    :param payload: Row mappings to upsert; the first mapping's keys decide
699        which columns are overwritten on conflict.
700    :param session_inst: Active session.
701    :param model: Table model being upserted.
702    :param pk_fields: Conflict-target columns. Defaults to ``["id"]``.
703    :return: ``(True, rows)`` where ``rows`` are the upserted model instances.
704    """
705    if not pk_fields:
706        pk_fields = ["id"]
707    stmnt = upsert(model).values(payload)
708    stmnt = stmnt.on_conflict_do_update(
709        index_elements=[getattr(model, x) for x in pk_fields],
710        set_={k: getattr(stmnt.excluded, k) for k in payload[0].keys()},
711    )
712    results = (
713        session_inst.scalars(
714            stmnt.returning(model),
715            execution_options={"populate_existing": True},
716        )
717    ).all()
718
719    session_inst.commit()
720    invalidate_written(session_inst, model)
721
722    return True, results
